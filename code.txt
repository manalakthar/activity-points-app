import pyperclip
import sys
password = {'email':'manal123',
'blog':'123456'
}
if len(sys.argv)<2:
    sys.exit("Usage: python pyperclip_clipboard.py <account>")
    account = sys.argv[1]
    password= password.get(account)

    print(f"Password for {account} copied to clipboard ")
